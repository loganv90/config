# config files

includes config files for:
- alacritty
- tmux
- nvim
- zsh

prerequisites:
- mise
- rg
- fd
- tree-sitter-cli

installation:
- ln -s pathToRepo/nvim ~/.config/nvim
- ln -s pathToRepo/tmux ~/.config/tmux
- ln -s pathToRepo/alacritty ~/.config/alacritty
- source pathToRepo/zsh/.zshrc
